RIVA Topbar Measurement Diagnostic

Instructions:

  1. Open the RIVA webapp in your browser (localhost:3001)
  2. Open DevTools Console (F12)
  3. Copy and paste the script below into the console
  4. Press Enter to run

Diagnostic Script:

// RIVA Topbar Measurement Diagnostic
// Copy this entire block and paste into browser console

(function() {
  console.log('=== RIVA Topbar Measurement ===\n');

  // Find the oval Glass container (Theme + Business)
  const ovalGlass = document.querySelector('header [class*="lg-glass"], header .lg-glass');

  // Find the Profile Menu button
  const profileMenu = document.querySelector('header button[aria-label*="Profile"]');

  if (!ovalGlass) {
    console.error('❌ Could not find oval Glass container');
    return;
  }

  if (!profileMenu) {
    console.error('❌ Could not find Profile Menu button');
    return;
  }

  // Measure actual rendered dimensions
  const ovalRect = ovalGlass.getBoundingClientRect();
  const profileRect = profileMenu.getBoundingClientRect();

  console.log('📏 Oval Glass (Theme + Business):');
  console.log(`   Width: ${ovalRect.width}px`);
  console.log(`   Height: ${ovalRect.height}px`);
  console.log(`   Top: ${ovalRect.top}px`);
  console.log(`   Bottom: ${ovalRect.bottom}px`);
  console.log(`   Vertical Center: ${(ovalRect.top + ovalRect.bottom) / 2}px`);

  console.log('\n🔵 Profile Button:');
  console.log(`   Width: ${profileRect.width}px`);
  console.log(`   Height: ${profileRect.height}px`);
  console.log(`   Top: ${profileRect.top}px`);
  console.log(`   Bottom: ${profileRect.bottom}px`);
  console.log(`   Vertical Center: ${(profileRect.top + profileRect.bottom) / 2}px`);

  console.log('\n📊 Comparison:');
  const heightDiff = Math.abs(ovalRect.height - profileRect.height);
  const centerDiff = Math.abs(
    (ovalRect.top + ovalRect.bottom) / 2 -
    (profileRect.top + profileRect.bottom) / 2
  );

  console.log(`   Height difference: ${heightDiff.toFixed(2)}px`);
  console.log(`   Vertical center difference: ${centerDiff.toFixed(2)}px`);

  if (heightDiff < 1 && centerDiff < 1) {
    console.log('\n✅ PERFECT: Both surfaces match in height and alignment!');
  } else {
    console.log('\n❌ MISMATCH DETECTED:');
    if (heightDiff >= 1) {
      console.log(`   - Profile button needs to be ${heightDiff.toFixed(2)}px ${profileRect.height < ovalRect.height ? 'larger' : 'smaller'}`);
    }
    if (centerDiff >= 1) {
      console.log(`   - Profile button is ${centerDiff.toFixed(2)}px ${profileRect.top < ovalRect.top ? 'higher' : 'lower'} than the oval`);
    }
  }

  console.log('\n🔍 Computed Styles:');
  const ovalStyles = window.getComputedStyle(ovalGlass);
  const profileStyles = window.getComputedStyle(profileMenu);

  console.log('Oval Glass:');
  console.log(`   height: ${ovalStyles.height}`);
  console.log(`   min-height: ${ovalStyles.minHeight}`);
  console.log(`   padding: ${ovalStyles.padding}`);

  console.log('Profile Button:');
  console.log(`   height: ${profileStyles.height}`);
  console.log(`   width: ${profileStyles.width}`);
  console.log(`   min-height: ${profileStyles.minHeight}`);
  console.log(`   min-width: ${profileStyles.minWidth}`);

  console.log('\n=== End Measurement ===');
})();

What to look for:

  • Both surfaces should have exactly 48px height
  • Profile button should be 48×48px (width = height = 48)
  • Height difference should be < 1px
  • Vertical center difference should be < 1px
